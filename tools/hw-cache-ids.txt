#   vendor = 4-digit hex Vendor ID
#   device = 4-digit hex Device ID
#
# Used by tools/build-hw-cache.ps1 to pre-fetch pages into lsl-hw-cache/.
# The installed tool (install.ps1) reads that folder so common hardware is
# rated with no per-device network request.

# rating: 8086:2723, 10EC:2600, 10DE:1E91, 144D:A808, 1BB1:5013, 04F3:0C11,
# 13D3:56A2, 8086:9BC4, 8087:0029.
